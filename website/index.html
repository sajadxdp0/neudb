from neudb.ai_schema import (
    add_message_with_embedding, add_user, create_session,
    embed_text, init_ai_database,
)

db = init_ai_database("my_memory")
alice = add_user(db, "alice", "alice@example.com")
session = create_session(db, alice, "Debugging Python")

add_message_with_embedding(
    db, session, "user", "How do I fix ModuleNotFoundError?"
)

messages = db.table("messages")
vector = embed_text("Python import errors")
if vector is None:
    results = messages.search_text("content", "Python import errors", top_k=5)
else:
    results = messages.search_similar("embedding", vector, top_k=5)

Install .[embeddings] or .[agent] for embeddings. First use downloads all-MiniLM-L6-v2; embed_text() returns None when the dependency is unavailable, so this example falls back to text search. Download or load errors are surfaced.

# Create and query tables
neudb --db ./neudb_data table create users
neudb --db ./neudb_data row insert users --data '{"username":"alice"}'
neudb --db ./neudb_data row list users --limit 10

# Vector search
neudb --db ./neudb_data row search messages \
  --field embedding \
  --vector '[0.9, 0.1, 0.2]'

# Local Ollama agent with memory
neudb-agent --provider ollama --model llama3.2 --db ./agent_memory
# Create a user
curl -X POST http://127.0.0.1:8000/users \
  -H "Content-Type: application/json" \
  -H "X-API-Key: $NEUDB_API_KEY" \
  -d '{"username":"alice","email":"alice@example.com"}'

# Query search with text fallback
curl -X POST http://127.0.0.1:8000/search \
  -H "Content-Type: application/json" \
  -H "X-API-Key: $NEUDB_API_KEY" \
  -d '{"query":"Python import error","top_k":5}'
